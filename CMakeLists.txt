cmake_minimum_required(VERSION 3.24)
project(memory_inspector LANGUAGES CXX)
include(FetchContent)
include(CTest)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(INSPECTOR_BENCH "Build Google Benchmark workloads" ON)
add_library(memory INTERFACE)
target_include_directories(memory INTERFACE include)
if(BUILD_TESTING)
    set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
    FetchContent_Declare(googletest URL https://codeload.github.com/google/googletest/tar.gz/refs/tags/v1.16.0)
    FetchContent_MakeAvailable(googletest)
    add_executable(memory_tests tests.cpp)
    target_link_libraries(memory_tests PRIVATE memory GTest::gtest_main)
    add_test(NAME memory_tests COMMAND memory_tests)
endif()
if(INSPECTOR_BENCH)
    set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(benchmark URL https://codeload.github.com/google/benchmark/tar.gz/refs/tags/v1.9.1)
    FetchContent_MakeAvailable(benchmark)
    add_executable(memory_bench bench.cpp)
    target_link_libraries(memory_bench PRIVATE memory benchmark::benchmark_main)
endif()
