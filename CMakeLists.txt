cmake_minimum_required(VERSION 3.24)
project(memory_inspector LANGUAGES CXX)
include(FetchContent)
include(CTest)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(INSPECTOR_GUI "Build the SDL3/Dear ImGui inspector" ON)
option(INSPECTOR_BENCH "Build Google Benchmark workloads" ON)
option(INSPECTOR_ASAN "Enable AddressSanitizer" OFF)
add_library(memory INTERFACE)
target_include_directories(memory INTERFACE include)
if(INSPECTOR_ASAN)
    if(MSVC)
        add_compile_options(/fsanitize=address)
        add_link_options(/INCREMENTAL:NO)
    else()
        target_compile_options(memory INTERFACE -fsanitize=address -fno-omit-frame-pointer)
        target_link_options(memory INTERFACE -fsanitize=address)
    endif()
endif()
if(BUILD_TESTING)
    set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
    FetchContent_Declare(googletest URL https://codeload.github.com/google/googletest/tar.gz/refs/tags/v1.16.0)
    FetchContent_MakeAvailable(googletest)
    add_executable(memory_tests tests.cpp)
    target_link_libraries(memory_tests PRIVATE memory GTest::gtest_main)
    add_test(NAME memory_tests COMMAND memory_tests)
endif()
if(INSPECTOR_BENCH)
    set(BENCHMARK_ENABLE_TESTING OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(benchmark URL https://codeload.github.com/google/benchmark/tar.gz/refs/tags/v1.9.1)
    FetchContent_MakeAvailable(benchmark)
    add_executable(memory_bench bench.cpp)
    target_link_libraries(memory_bench PRIVATE memory benchmark::benchmark_main)
endif()
if(INSPECTOR_GUI)
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    FetchContent_Declare(SDL3 URL https://codeload.github.com/libsdl-org/SDL/tar.gz/refs/tags/release-3.2.10)
    FetchContent_Declare(imgui URL https://codeload.github.com/ocornut/imgui/tar.gz/refs/tags/v1.91.9b)
    FetchContent_MakeAvailable(SDL3 imgui)
    add_executable(inspector demo.cpp
        ${imgui_SOURCE_DIR}/imgui.cpp ${imgui_SOURCE_DIR}/imgui_draw.cpp
        ${imgui_SOURCE_DIR}/imgui_tables.cpp ${imgui_SOURCE_DIR}/imgui_widgets.cpp
        ${imgui_SOURCE_DIR}/backends/imgui_impl_sdl3.cpp
        ${imgui_SOURCE_DIR}/backends/imgui_impl_sdlrenderer3.cpp)
    target_include_directories(inspector PRIVATE ${imgui_SOURCE_DIR})
    target_link_libraries(inspector PRIVATE memory SDL3::SDL3)
endif()
